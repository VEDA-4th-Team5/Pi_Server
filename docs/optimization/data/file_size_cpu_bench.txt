  W=$(mktemp -d); mkdir -p $W/include/ocr $W/scene
  git show d41e2f09:src/ocr/PlateImageEnhancer.cpp     > $W/PlateImageEnhancer.cpp
  git show d41e2f09:include/ocr/PlateImageEnhancer.hpp > $W/include/ocr/PlateImageEnhancer.hpp

  cp docs/optimization/data/file_size_cpu_stages.cpp $W/stages.cpp
  cd $W && g++ -O2 -std=c++20 -w -I include bench.cpp  PlateImageEnhancer.cpp $(pkg-config --cflags --libs opencv4) -o bench
           g++ -O2 -std=c++20 -w -I include stages.cpp PlateImageEnhancer.cpp $(pkg-config --cflags --libs opencv4) -o stages
  ./bench 15 1     # 1스레드 총시간
